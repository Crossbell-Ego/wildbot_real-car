# 啟動底盤與硬體驅動服務 (主機端 Terminal)
start

# 執行手臂位置教導錄製器 (主機端 Terminal，或打原指令進容器執行)
teach
# 備用原指令: docker exec -it -w /workspaces/workspaces/src/arm_ik/arm_ik/ compose-kros_car-1 python3 grab_teach.py

# 執行手臂自動化任務 (單點或動作序列，例如回放 1 號點)
arm 1
arm 1 2 1
# 備用原指令: docker exec -it -w /workspaces/workspaces/src/arm_ik/arm_ik/ compose-kros_car-1 python3 grab_execute.py 1 2 1

# 執行參數校正與量測工具
calib
# 網頁版儀表板網址：http://localhost:8080 或 http://<小車IP>:8080

# 1. 啟動 Nav2 導航系統 (主機端 Terminal)
# 載入已儲存的地圖進行導航
nstart
# 或是邊建圖邊導航 (SLAM + Nav2)
nstart slam

# 2. 初始化定位 (主機端 Terminal)
initpose

# 3. 執行導航座標教導/回放工具 (主機端 Terminal)
nteach
# 備用原指令: docker exec -it -w /workspaces/workspaces/src/arm_ik/arm_ik/ compose-kros_car-1 python3 nav_teach.py

# 4. 執行導航路徑自動化任務 (主機端 Terminal)
nav 1
nav 1 2
# 備用原指令: docker exec -it -w /workspaces/workspaces/src/arm_ik/arm_ik/ compose-kros_car-1 python3 nav_execute.py 1 2
